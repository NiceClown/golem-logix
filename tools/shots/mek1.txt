# Mekanism fluids with the Golem Interface (gradle -Pmekanism): brine kept by a bucket slot and pulled out by a Mekanism pipe
# into a Mekanism tank; heavy water pulled from another tank by a pipe and poured into the interface -> network
reload
cmd time set day
cmd kill @e[type=!player]
cmd fill -40 -61 -20 40 -61 14 minecraft:grass_block
cmd fill -40 -60 -20 40 -45 14 minecraft:air
hook rig 0 -60 0
cmd setblock 10 -60 -2 golem_logix:golem_interface
# output: pipe north of the interface pulls from it (south side = PULL) into a basic fluid tank
cmd setblock 10 -60 -3 mekanism:basic_mechanical_pipe
cmd setblock 10 -60 -4 mekanism:basic_fluid_tank
# intake: tank above the interface (Mekanism tanks give fluid from their bottom), pipe between pulls from it (up side = PULL)
# and pushes down into the interface
cmd setblock 10 -58 -2 mekanism:basic_fluid_tank
cmd setblock 10 -59 -2 mekanism:basic_mechanical_pipe
wait 40
cmd data merge block 10 -60 -3 {connection:[I;0,0,0,2,0,0]}
cmd data merge block 10 -59 -2 {connection:[I;0,2,0,0,0,0]}
wait 60
hook netfluidput 0 -60 0 mekanism:brine 6000
hook iface 10 -60 -2 0 mekanism:brine_bucket 4
hook fillfluid 10 -58 -2 mekanism:heavy_water 6000
wait 200
hook fluidcheck 10 -60 -4 mekanism:brine 2000
hook fluidcheck 10 -58 -2 mekanism:heavy_water 0
hook nbt 10 -59 -2
hook fluidcheck 10 -60 -2 mekanism:heavy_water 0
hook netfluidcheck 0 -60 0 mekanism:heavy_water 5900
look 10.5 -59.6 -2.5 3.4 135 25
wait 5
shot mek_pipes
look 10.5 -59.5 -1.5 2.5 180 20
wait 5
use 10 -60 -2
wait 10
shot mek_gui
closeScreen
