# JEI precraft: Ctrl + "+" orders the missing planks and sticks of a wooden pickaxe, then "+" lays the recipe out (needs: start -Mods)
reload
cmd time set day
cmd kill @e[type=item]
cmd fill -40 -61 -14 40 -61 10 minecraft:deepslate_tiles
cmd fill -40 -60 -14 40 -50 10 minecraft:air
cmd kill @e[type=item]
hook rig 0 -60 0
cmd setblock 3 -60 -2 golem_logix:crafting_terminal[facing=north]
wait 120
hook netput 0 -60 0 minecraft:oak_log 8
look 3.5 -59.5 -1.5 3 0 8
use 3 -60 -2
wait 40
jeitransfer minecraft:wooden_pickaxe COSMETIC ctrl
wait 10
shot pre_ordered
gui on
wait 60
shot pre_toast
gui off
hook netcheck 0 -60 0 minecraft:oak_log 6
jeitransfer minecraft:wooden_pickaxe ok
wait 20
shot pre_filled
closeScreen
